package com.wechatai.tool.impl;

        + "示例：1523*0.85、100美元等于多少人民币、5英尺是多少厘米")
    public String calculate(
        @P("计算表达式或换算请求。支持：数学表达式(1523*0.85)、"
            + "单位换算(5英尺=多少厘米)、汇率换算(100美元=多少人民币)")
        String expression) {

        expression = expression.trim();

        // 1. 汇率换算（走实时 API）
        if (expression.contains("美元") || expression.contains("人民币")

            ScriptEngine engine = new ScriptEngineManager().getEngineByName("JavaScript");
            // 安全处理：只允许数学表达式

            }

            sanitized = sanitized.replace("^", "**");
            Object result = engine.eval(sanitized);

            return String.format("%s = %s", expression, formatNumber(result));
        } catch (Exception e) {
            return "无法计算该表达式，请检查格式。示例：1523 * 0.85、100 + 200 * 3";
        }

### 4.2 `translate` — 翻译

```java
package com.wechatai.tool.impl;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import dev.langchain4j.agent.tool.P;
import dev.langchain4j.agent.tool.Tool;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;

import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.Map;

/**
 * 多语言翻译工具（READ）— 中英日韩法等语言互译。
 * 对接百度翻译 API，自动检测源语言。
 */
@Component
public class TranslateTool {

    private static final String BAIDU_API = "https://api.fanyi.baidu.com/api/trans/vip/translate";

    private static final Map<String, String> LANG_CODE = Map.of(
        "中文", "zh", "英语", "en", "日语", "jp",
        "韩语", "kor", "法语", "fra", "德语", "de",
        "西班牙语", "spa", "俄语", "ru"
    );

    @Value("${translate.baidu.app-id}")
    private String appId;

    @Value("${translate.baidu.secret-key}")
    private String secretKey;

    private final RestClient restClient;
    private final ObjectMapper objectMapper;

    public TranslateTool(RestClient.Builder restClientBuilder) {
        this.restClient = restClientBuilder.build();
        this.objectMapper = new ObjectMapper();
    }

    @Tool("多语言翻译工具，支持中文、英语、日语、韩语、法语、德语等常见语言互译。"
        + "自动检测源语言，不需要用户指定从什么语言翻译。"
        + "当用户说「翻译xxx」「xxx用英语怎么说」「这句话是什么意思」时使用")
    public String translate(
        @P("需要翻译的文本内容") String text,
        @P("目标语言，如：英语、日语、法语。不传则自动判断语言并翻译为目标语言")
        String targetLang) {

        if (text == null || text.trim().isEmpty()) {
            return "请提供需要翻译的内容";
        }

        String targetCode = LANG_CODE.getOrDefault(targetLang, "en");

        try {
            String salt = String.valueOf(System.currentTimeMillis());
            String sign = md5(appId + text + salt + secretKey);

            String response = restClient.get()
                .uri(uriBuilder -> uriBuilder
                    .scheme("https").host("api.fanyi.baidu.com")
                    .path("/api/trans/vip/translate")
                    .queryParam("q", text)
                    .queryParam("from", "auto")
                    .queryParam("to", targetCode)
                    .queryParam("appid", appId)
                    .queryParam("salt", salt)
                    .queryParam("sign", sign)
                    .build())
                .retrieve()
                .body(String.class);

            JsonNode root = objectMapper.readTree(response);
            JsonNode transResult = root.path("trans_result");

            if (transResult.isArray() && transResult.size() > 0) {
                String src = transResult.get(0).path("src").asText();
                String dst = transResult.get(0).path("dst").asText();

                return String.format(
                    "【翻译结果】\n原文：%s\n译文：%s",
                    src, dst);
            }

            return "翻译失败：" + root.path("error_msg").asText("未知错误");
        } catch (Exception e) {
            return "翻译服务暂时不可用：" + e.getMessage();
        }
    }

    private String md5(String input) {
        try {
            MessageDigest md = MessageDigest.getInstance("MD5");
            byte[] digest = md.digest(input.getBytes());
            StringBuilder sb = new StringBuilder();
            for (byte b : digest) {
                sb.append(String.format("%02x", b));
            }
            return sb.toString();
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException(e);
        }
    }
}
```

**实现要点**：
- 必填配置：`translate.baidu.app-id` 和 `translate.baidu.secret-key`
- 注册地址：`https://api.fanyi.baidu.com` — 个人开发者免费 100 万字符/月
- `from=auto` 自动检测源语言
- 需要 MD5 签名（Java 标准库 `MessageDigest`）

### 4.3 `queryNews` — 新闻搜索

```java
package com.wechatai.tool.impl;

import dev.langchain4j.agent.tool.P;
import dev.langchain4j.agent.tool.Tool;
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.springframework.stereotype.Component;

import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;
import java.time.LocalDate;

/**
 * 新闻搜索工具（READ）— 搜索最新新闻资讯。
 * 用 Jsoup 抓百度新闻结果，零外部依赖。
 */
@Component
public class NewsTool {

    @Tool("搜索最新的新闻资讯。当用户询问时事、热点新闻、行业动态时使用。"
        + "返回新闻标题、来源、发布时间和链接。多条结果按时间倒序排列")
    public String queryNews(
        @P("搜索关键词，如：人工智能 最新、NBA 赛果、A股 行情") String keyword,
        @P("返回结果数量，默认5条，最多10条") Integer maxResults) {

        if (keyword == null || keyword.trim().isEmpty()) {
            return "请提供搜索关键词";
        }

        int limit = maxResults != null ? Math.min(maxResults, 10) : 5;

        try {
            String url = "https://news.baidu.com/ns?word="
                + URLEncoder.encode(keyword, StandardCharsets.UTF_8)
                + "&pn=0&rn=" + limit;

            Document doc = Jsoup.connect(url)
                .userAgent("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36")
                .timeout(15000)
                .execute()
                .parse();

            StringBuilder sb = new StringBuilder();
            sb.append("🔍 ").append(keyword).append(" 最新新闻\n\n");

            int count = 0;
            for (Element result : doc.select(".result")) {
                if (count >= limit) break;

                Element titleEl = result.selectFirst("h3 a");
                Element sourceEl = result.selectFirst(".c-author");
                Element abstractEl = result.selectFirst(".c-summary");

                if (titleEl == null) continue;

                String title = titleEl.text().strip();
                String link = titleEl.attr("href");
                String source = sourceEl != null ? sourceEl.text().strip() : "";
                String summary = abstractEl != null ? abstractEl.text().strip() : "";

                sb.append("📰 ").append(title).append("\n");
                sb.append("   来源：").append(source).append("\n");
                sb.append("   摘要：").append(summary).append("\n\n");

                count++;
            }

            if (count == 0) {
                return "未找到 " + keyword + " 的相关新闻，请更换关键词试试";
            }

            return sb.toString();
        } catch (Exception e) {
            return "新闻搜索失败：" + e.getMessage();
        }
    }
}
```

**实现要点**：
- 纯 Jsoup 抓取，已有依赖
- 备选入口：必应新闻 `https://cn.bing.com/news/search?q=`
- 返回格式：标题 + 来源 + 摘要 + 时间
- 多个结果按相关性排序（搜索引擎已排好）

### 4.4 `queryDictionary` — 词典/百科

```java
package com.wechatai.tool.impl;

import dev.langchain4j.agent.tool.P;
import dev.langchain4j.agent.tool.Tool;
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.springframework.stereotype.Component;

import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;

/**
 * 词典百科工具（READ）— 查询词语解释或百科知识。
 * 自动判断查询类型：单词释义、成语典故、百科知识。
 */
@Component
public class DictionaryTool {

    @Tool("查询词语的解释、用法、例句或百科知识。"
        + "当用户问一个词是什么意思、成语典故、百科知识时使用。"
        + "自动判断查询类型：单词(查词典)、成语(查出处)、百科(查知识)")
    public String queryDictionary(
        @P("要查询的词语或百科条目名称") String word,
        @P("查询类型：WORD(词典释义)/IDIOM(成语出处)/ENCYCLOPEDIA(百科知识)"
            + "，不填则自动判断") String type) {

        if (word == null || word.trim().isEmpty()) {
            return "请提供要查询的词语";
        }

        // 简化处理：优先调百度百科
        return queryBaike(word);
    }

    private String queryBaike(String word) {
        try {
            String url = "https://baike.baidu.com/item/"
                + URLEncoder.encode(word, StandardCharsets.UTF_8);

            Document doc = Jsoup.connect(url)
                .userAgent("Mozilla/5.0")
                .timeout(10000)
                .execute()
                .parse();

            Element summary = doc.selectFirst(".lemma-summary");
            Element title = doc.selectFirst("h1");

            if (title == null) {
                return "未找到「" + word + "」的相关信息";
            }

            String titleText = title.text().strip();
            String summaryText = summary != null ? summary.text().strip() : "(暂无摘要)";

            return String.format(
                "【%s】\n%s\n\n（来源：百度百科）",
                titleText, summaryText);
        } catch (Exception e) {
            return "查询失败：" + e.getMessage();
        }
    }
}
```

**实现要点**：
- 主入口：Jsoup 抓百度百科 `https://baike.baidu.com/item/xxx`
- 备选：词典专用 API（百度汉语、有道词典）
- 支持：单词释义、成语典故、百科知识
